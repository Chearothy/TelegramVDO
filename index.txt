  function validId(v){return typeof v==='string'&&/^[A-Za-z0-9_-]{5,200}$/.test(v);}

  function extractFolderId(value){
    const link=String(value||'').trim();
    if(!link) throw new Error('Paste a Google Drive folder link.');
    const match = link.match(/\/folders\/([A-Za-z0-9_-]+)/);
    if(match && validId(match[1])) return match[1];
    if(validId(link)) return link;
    throw new Error('Use a valid Google Drive folder link.');
  }

  function validateFolderLink(value){
    const id = extractFolderId(value);
    return 'https://drive.google.com/drive/folders/' + id;
  }

  async function listFolderPage(folderId, pageToken, pageSize){
    if (!GOOGLE_API_KEY) {
      throw new Error('Google Drive API key is missing.');
    }
    const q = encodeURIComponent(`'${folderId}' in parents and trashed = false`);
    let url = `https://www.googleapis.com/drive/v3/files?q=${q}&fields=nextPageToken,files(id,name,mimeType,modifiedTime,webViewLink)&pageSize=${pageSize||100}&key=${GOOGLE_API_KEY}`;
    if (pageToken) url += `&pageToken=${encodeURIComponent(pageToken)}`;

    const res = await fetch(url);
    if (!res.ok) {
      const err = await res.json().catch(() => ({}));
      throw new Error((err.error && err.error.message) || 'Failed to fetch contents from Google Drive.');
    }
    return await res.json();
  }

  async function readFolders(){
    const list = loadLocalFolders();
    return list.sort((a,b) => a.name.localeCompare(b.name));
  }

  function linkForChild(file){
    return 'https://drive.google.com/drive/folders/' + encodeURIComponent(file.id);
  }

  async function scanFolder(folder){
    let token = null, pages = 0, subfolders = [], videos = [];
    const folderId = extractFolderId(folder.link || folder.id);

    do {
      if (pages >= 30) throw new Error('Folder contains too many items (exceeds 3,000).');
      const r = await listFolderPage(folderId, token, 100);
      (r.files || []).forEach(f => {
        if (!f || !validId(f.id) || typeof f.mimeType !== 'string') return;
        const name = String(f.name || 'Untitled');
        if (f.mimeType === 'application/vnd.google-apps.folder') {
          subfolders.push({ id: f.id, name: name, link: linkForChild(f) });
        } else if (f.mimeType.toLowerCase().startsWith('video/')) {
          videos.push({ id: f.id, name: name, modifiedTime: f.modifiedTime || '', folderLabel: folder.name || 'Drive folder' });
        }
      });
      token = r.nextPageToken || null;
      pages++;
      if (subfolders.length + videos.length > MAX_ITEMS) {
        throw new Error('This folder has more than 3,000 items.');
      }
    } while(token);

    subfolders.sort((a,b) => a.name.localeCompare(b.name));
    videos.sort((a,b) => a.name.localeCompare(b.name));
    return { subfolders: subfolders, videos: videos };
  }

  async function refreshCurrent(silent){
    if(busy) return;
    busy = true;
    const btn = document.getElementById('refresh-button');
    btn.disabled = true;
    document.getElementById('folder-grid').setAttribute('aria-busy', 'true');
    syncStatus.className = 'sync-status';
    syncStatus.textContent = path.length ? 'Loading ' + path[path.length - 1].name + '…' : 'Loading saved folders…';

    try {
      folders = await readFolders();
      renderSettings();

      if(!path.length){
        currentFolders = [];
        currentVideos = [];
        renderBrowse();
        syncStatus.textContent = folders.length ? 'Showing ' + folders.length + ' public Drive folder' + (folders.length === 1 ? '' : 's') + '.' : 'No folders connected. Add one in Settings.';
        syncStatus.className = 'sync-status ' + (folders.length ? 'ok' : '');
        lastSync.textContent = 'Open a folder to load its contents';
        return;
      }

      const current = path[path.length - 1];
      const data = await scanFolder(current);
      currentFolders = data.subfolders;
      currentVideos = data.videos;
      renderBrowse();

      const t = new Date().toLocaleTimeString([], { hour: 'numeric', minute: '2-digit' });
      syncStatus.textContent = 'Loaded ' + currentFolders.length + ' subfolder' + (currentFolders.length === 1 ? '' : 's') + ' and ' + currentVideos.length + ' video' + (currentVideos.length === 1 ? '' : 's') + '.';
      syncStatus.className = 'sync-status ok';
      lastSync.textContent = 'Last checked ' + t + ' · refreshes every 5 minutes';
    } catch(error) {
      folderGrid.replaceChildren();
      videoGrid.replaceChildren();
      emptyState.hidden = true;
      searchEmpty.hidden = true;
      syncStatus.className = 'sync-status error';
      syncStatus.textContent = 'Could not load folder: ' + error.message;
      lastSync.textContent = 'No current folder listing';
    } finally {
      busy = false;
      btn.disabled = false;
      document.getElementById('folder-grid').removeAttribute('aria-busy');
    }
  }

  async function addFolder(){
    const linkInput = document.getElementById('new-link'), labelInput = document.getElementById('new-label');
    addButton.disabled = true;
    setMessage(addStatus, 'Checking folder link…', '');

    try {
      const link = validateFolderLink(linkInput.value);
      const folderId = extractFolderId(link);
      const key = 'folders/' + folderId;
      const list = loadLocalFolders();

      if(list.some(f => f.key === key)){
        setMessage(addStatus, 'That folder is already added.', 'error');
        return;
      }

      const label = labelInput.value.trim().slice(0, 80) || 'Public Drive folder';
      list.push({ key: key, id: folderId, name: label, link: link });
      saveLocalFolders(list);

      setMessage(addStatus, 'Folder added successfully.', 'ok');
      labelInput.value = '';
      linkInput.value = '';
      path = [];
      await refreshCurrent(false);
    } catch(error) {
      setMessage(addStatus, error.message || 'Could not add folder.', 'error');
    } finally {
      addButton.disabled = false;
    }
  }

  async function saveEdit(){
    const save = document.getElementById('save-edit');
    save.disabled = true;
    setMessage(editStatus, 'Saving changes…', '');

    try {
      const newLink = validateFolderLink(document.getElementById('edit-link').value);
      const newId = extractFolderId(newLink);
      const newKey = 'folders/' + newId;
      const label = document.getElementById('edit-label').value.trim().slice(0, 80) || 'Public Drive folder';
      let list = loadLocalFolders();

      const idx = list.findIndex(f => f.key === editingKey);
      if(idx === -1) throw new Error('Folder entry not found.');

      list[idx] = { key: newKey, id: newId, name: label, link: newLink };
      saveLocalFolders(list);

      closeEdit();
      path = [];
      await refreshCurrent(false);
    } catch(error) {
      setMessage(editStatus, error.message || 'Could not save folder changes.', 'error');
    } finally {
      save.disabled = false;
    }
  }

  async function confirmRemove(f, button){
    if(button.dataset.armed !== 'yes'){
      button.dataset.armed = 'yes';
      button.textContent = 'Confirm remove';
      return;
    }
    let list = loadLocalFolders();
    list = list.filter(item => item.key !== f.key);
    saveLocalFolders(list);
    path = [];
    await refreshCurrent(false);
  }

  // Autoplay: the video that is mostly on screen plays (muted, as browsers require until the
  // viewer unmutes once; after that later videos play with sound). Only one plays at a time.
  function playTele(vid){
    teleGrid.querySelectorAll('video').forEach(function(o){ if(o!==vid && !o.paused) o.pause(); });
    vid.muted=!teleSound;
    const p=vid.play();
    if(p && p.catch) p.catch(function(){ vid.muted=true; vid.play().catch(function(){}); });
  }
  const teleObserver=('IntersectionObserver' in window) ? new IntersectionObserver(function(entries){
    entries.forEach(function(en){
      if(en.isIntersecting && en.intersectionRatio>=0.6) playTele(en.target);
      else if(!en.target.paused) en.target.pause();
    });
  },{threshold:[0,0.6,1]}) : null;

  function teleFiltered(){
    const q=teleSearch.value.trim().toLowerCase();
    if(!q) return teleAll;
    return teleAll.filter(function(v){ return ((v.title||'')+' '+(v.text||'')+' #'+v.id).toLowerCase().indexOf(q)!==-1; });
  }

  function makeTeleCard(v){
    const card=document.createElement('article'); card.className='video-card';
    const head=document.createElement('div'); head.className='video-head';
    const title=document.createElement('h3'); title.className='video-title'; title.textContent=v.title||('Video #'+v.id);
    const badge=document.createElement('span'); badge.className='folder-badge'; badge.textContent='@'+TELEGRAM_CHANNEL;
    head.append(title,badge);

    const srcUrl=teleUrl(v.stream||v.src);
    let frame;
    if(srcUrl){
      frame=document.createElement('div'); frame.className='video-frame';
      const video=document.createElement('video');
      video.controls=true; video.preload='none'; video.playsInline=true; video.muted=true;
      video.setAttribute('referrerpolicy','no-referrer');
      if(v.thumb) video.poster=teleUrl(v.thumb);
      video.src=srcUrl;
      video.addEventListener('volumechange', function(){ if(!video.paused) teleSound=!video.muted; });
      video.addEventListener('ended', function(){ // go on to the next video
        const next=card.nextElementSibling;
        if(next){ next.scrollIntoView({behavior:'smooth',block:'center'}); }
        else if(!teleMore.hidden){ renderTele(false); const n=card.nextElementSibling; if(n) n.scrollIntoView({behavior:'smooth',block:'center'}); }
      });
      video.addEventListener('error', function(){ // blocked or expired: try Telegram's own player
        if(frame.parentNode) frame.replaceWith(teleEmbedFrame(v.id));
      }, {once:true});
      frame.appendChild(video);
      if(teleObserver) teleObserver.observe(video);
    } else {
      frame=teleEmbedFrame(v.id); // no direct link available: use Telegram's own player
    }

    const meta=document.createElement('div'); meta.className='video-meta';
    const left=document.createElement('span'); left.textContent='Streams from Telegram'+(v.duration?' · '+v.duration:'')+(v.size?' · '+(v.size>=1073741824?(v.size/1073741824).toFixed(1)+' GB':Math.max(1,Math.round(v.size/1048576))+' MB'):'');
    const right=document.createElement('span');
    const d=v.date?new Date(v.date):null;
    right.textContent=(d&&!isNaN(d))?d.toLocaleDateString():('#'+v.id);
    meta.append(left,right);
    card.append(head,frame,meta);
    return card;
  }

  function renderTele(reset){
    if(reset){ teleGrid.replaceChildren(); teleShown=0; }
    const list=teleFiltered();
    const next=list.slice(teleShown,teleShown+TELE_PAGE);
    next.forEach(function(v){ teleGrid.appendChild(makeTeleCard(v)); });
    teleShown+=next.length;
    teleMore.hidden=teleShown>=list.length;
    teleEmpty.hidden=list.length>0;
  }

  async function loadTele(){
    setMessage(teleStatus,'Loading videos…','');
    try{
      const r=await fetch(TELE_API_BASE+'/api/videos',{cache:'no-store'});
      if(!r.ok) throw new Error('server answered '+r.status);
      const d=await r.json();
      teleAll=Array.isArray(d.videos)?d.videos:[];
      teleLoadedAt=Date.now();
      let msg=teleAll.length+' video'+(teleAll.length===1?'':'s');
      if(d.loading) msg+=' so far, still scanning the channel. Press Refresh in a minute.';
      if(d.error) msg+=' (last update problem: '+d.error+')';
      setMessage(teleStatus,msg,d.error?'error':'ok');
      renderTele(true);
    }catch(e){
      setMessage(teleStatus,'Cannot reach the video server ('+e.message+'). Start server.js, or set TELE_API_BASE.','error');
    }
  }

  function openTele(){ if(!teleLoadedAt || Date.now()-teleLoadedAt>120000) loadTele(); }

  // Telegram's player tells the page how tall it is; resize the matching iframe.
  window.addEventListener('message', function(e){
    if(e.origin!=='https://t.me') return;
    let d; try{ d=typeof e.data==='string'?JSON.parse(e.data):e.data; }catch(err){ return; }
    if(!d||d.event!=='resize'||!d.height) return;
    const frames=teleGrid.querySelectorAll('iframe');
    for(let i=0;i<frames.length;i++){
      if(frames[i].contentWindow===e.source){ frames[i].style.height=Math.max(80,Math.min(Number(d.height)||0,1400))+'px'; break; }
    }
  });

  function csvEscape(value){
    const text = String(value == null ? '' : value);
    return /[",\r\n]/.test(text) ? '"' + text.replace(/"/g, '""') + '"' : text;
  }

  function parseCsv(text){
    const rows = [];
    let row = [], field = '', quoted = false;
    text = String(text || '').replace(/^\uFEFF/, '');

    // Support CSV files exported by an earlier version that wrote escape
    // sequences such as "\uFEFF" and "\r\n" literally.
    if(text.startsWith('\\uFEFF')) text = text.slice(6);
    if(text.includes('\\r\\n') && !text.includes('\r\n')) {
      text = text.replace(/\\r\\n/g, '\n').replace(/\\n/g, '\n').replace(/\\r/g, '\n');
    }

    for(let i = 0; i < text.length; i++){
      const ch = text[i];
      if(quoted){
        if(ch === '"' && text[i + 1] === '"'){ field += '"'; i++; }
        else if(ch === '"') quoted = false;
        else field += ch;
      } else if(ch === '"' && field === ''){
        quoted = true;
      } else if(ch === ','){
        row.push(field); field = '';
      } else if(ch === '\n' || ch === '\r'){
        if(ch === '\r' && text[i + 1] === '\n') i++;
        row.push(field); rows.push(row); row = []; field = '';
      } else {
        field += ch;
      }
    }
    if(quoted) throw new Error('CSV contains an unclosed quoted field.');
    if(field !== '' || row.length){
      row.push(field); rows.push(row);
    }
    return rows.filter(r => r.some(cell => String(cell).trim() !== ''));
  }

  function exportCsv(){
    try {
      const list = loadLocalFolders();
      const rows = [['name', 'link', 'id', 'key']];
      list.forEach(f => rows.push([f.name || '', f.link || '', f.id || '', f.key || '']));
      const csv = '\uFEFF' + rows.map(row => row.map(csvEscape).join(',')).join('\r\n');
      const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = 'drive-folders-' + new Date().toISOString().slice(0, 10) + '.csv';
      document.body.appendChild(a);
      a.click();
      a.remove();
      URL.revokeObjectURL(url);
      setMessage(csvStatus, 'Exported ' + list.length + ' folder' + (list.length === 1 ? '' : 's') + ' to CSV.', 'ok');
    } catch(error) {
      setMessage(csvStatus, error.message || 'Could not export CSV.', 'error');
    }
  }

  async function importCsvFile(file){
    if(!file) return;
    importCsvButton.disabled = true;
    setMessage(csvStatus, 'Reading CSV file…', '');
    try {
      const rows = parseCsv(await file.text());
      if(!rows.length) throw new Error('The CSV file is empty.');
      const headers = rows.shift().map(h => h.trim().replace(/^\uFEFF/, '').toLowerCase());
      const nameIndex = headers.indexOf('name');
      const linkIndex = headers.indexOf('link');
      const idIndex = headers.indexOf('id');
      if(linkIndex === -1 && idIndex === -1) {
        throw new Error('CSV must include a "link" or "id" column. Expected columns: name,link.');
      }

      const list = loadLocalFolders();
      const knownIds = new Set(list.map(f => {
        try { return extractFolderId(f.link || f.id); } catch(e) { return ''; }
      }));
      let added = 0, skipped = 0, invalid = 0;
      rows.forEach(row => {
        try {
          const rawLink = linkIndex >= 0 ? (row[linkIndex] || '').trim() : '';
          const rawId = idIndex >= 0 ? (row[idIndex] || '').trim() : '';
          const id = extractFolderId(rawLink || rawId);
          if(knownIds.has(id)){ skipped++; return; }
          const name = ((nameIndex >= 0 ? row[nameIndex] : '') || '').trim().slice(0, 80) || 'Public Drive folder';
          const link = 'https://drive.google.com/drive/folders/' + id;
          list.push({ key: 'folders/' + id, id: id, name: name, link: link });
          knownIds.add(id);
          added++;
        } catch(e) { invalid++; }
      });
      if(added === 0 && invalid > 0) {
        throw new Error('No valid folders found. Check the CSV columns and Drive links.');
      }
      saveLocalFolders(list);
      path = [];
      await refreshCurrent(false);
      setMessage(csvStatus, 'Import complete: ' + added + ' added, ' + skipped + ' duplicate(s) skipped, ' + invalid + ' invalid row(s) skipped.', added ? 'ok' : '');
    } catch(error) {
      setMessage(csvStatus, error.message || 'Could not import CSV.', 'error');
    } finally {
      importCsvButton.disabled = false;
      csvFilePicker.value = '';
    }
  }
